Total Attempts vs Goals # ------------------------------------- sns.scatterplot( data=df, x='total attempts team1', y='number of goals team1', color='darkorange', s=90, alpha=0.75, edgecolor='black', ax=axes[1,0] ) sns.regplot( data=df, x='total attempts team1', y='number of goals team1', scatter=False, color='red', ax=axes[1,0] ) axes[1,0].set_title( 'Total Attempts vs Goals Scored', fontsize=13, fontweight='bold' ) axes[1,0].grid(alpha=0.3) # ------------------------------------- # 4. On Target Attempts vs Goals # ------------------------------------- sns.scatterplot( data=df, x='on target attempts team1', y='number of goals team1', color='purple', s=90, alpha=0.75, edgecolor='black', ax=axes[1,1] ) sns.regplot( data=df, x='on target attempts team1', y='number of goals team1', scatter=False, color='black', ax=axes[1,1] ) axes[1,1].set_title( 'On Target Attempts vs Goals Scored', fontsize=13, fontweight='bold' ) axes[1,1].grid(alpha=0.3) plt.tight_layout() plt.savefig("figures/figure_02.png", dpi=150, bbox_inches="tight") plt.show() # ===================================== # CANVAS 3 : DISCIPLINE & MATCH INTENSITY # ===================================== fig, axes = plt.subplots(2, 2, figsize=(18, 12)) fig.suptitle( 'Discipline and Match Intensity Analysis', fontsize=20, fontweight='bold' ) # ------------------------------------- # 1.
